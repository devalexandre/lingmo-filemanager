cmake_minimum_required(VERSION 3.16)
project(archive-tests LANGUAGES CXX)
set(CMAKE_AUTOMOC ON)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Test)
find_package(KF6KIO REQUIRED)
enable_testing()
add_executable(archive-test archivetest.cpp ../helper/archivemanager.cpp)
target_link_libraries(archive-test PRIVATE Qt6::Core Qt6::Test)
add_test(NAME archives COMMAND archive-test)

add_executable(browser-test browsertest.cpp ../model/searchmodel.cpp ../model/pathbarmodel.cpp ../helper/datehelper.cpp
               ../thumbnailer/thumbnailcache.cpp)
target_include_directories(browser-test PRIVATE ..)
target_link_libraries(browser-test PRIVATE Qt6::Core Qt6::Gui Qt6::Test KF6::KIOCore)
add_test(NAME browser COMMAND browser-test)

# Shared freedesktop.org thumbnail cache (same rules as Lingmo Photos)
add_executable(thumbnail-test thumbnailtest.cpp ../thumbnailer/thumbnailcache.cpp)
target_include_directories(thumbnail-test PRIVATE ..)
target_link_libraries(thumbnail-test PRIVATE Qt6::Core Qt6::Gui Qt6::Test)
add_test(NAME thumbnails COMMAND thumbnail-test)
set_tests_properties(thumbnails PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
